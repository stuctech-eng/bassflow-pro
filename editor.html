const canvas = document.getElementById("canvas"); const ctx = canvas.getContext("2d"); // Responsive canvas canvas.width = window.innerWidth; canvas.height = 320; let notes = []; let ghost = null; let isTouching = false; // Instellingen const step = 12; const staffTop = 80; // Iets lager voor de hoge noten const noteMap = ["b/5","a/5","g/5","f/5","e/5","d/5","c/5","b/4","a/4","g/4","f/4","e/4","d/4","c/4","b/3","a/3","g/3","f/3","e/3"]; // De 5 standaard lijnen van de notenbalk (index 4 t/m 12 in de noteMap voor een G-sleutel setup) const staffLineIndexes = [4, 6, 8, 10, 12]; function drawStaff() { ctx.strokeStyle = "#888"; ctx.lineWidth = 1; staffLineIndexes.forEach(index => { let y = staffTop + (index * (step / 2)); ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke(); }); } function getNote(x, y) { let index = Math.round((y - staffTop) / (step / 2)); index = Math.max(0, Math.min(noteMap.length - 1, index)); return { x: x, y: staffTop + index * (step / 2), note: noteMap[index], index: index }; } function drawLedgerLines(x, y, index) { ctx.strokeStyle = "white"; ctx.lineWidth = 2; const halfStep = step / 2; // Hulplijnen boven (index < 4) if (index < 4) { for (let i = 4 - 2; i >= index; i -= 2) { let ly = staffTop + (i * halfStep); ctx.beginPath(); ctx.moveTo(x - 15, ly); ctx.lineTo(x + 15, ly); ctx.stroke(); } } // Hulplijnen onder (index > 12) if (index > 12) { for (let i = 12 + 2; i <= index; i += 2) { let ly = staffTop + (i * halfStep); ctx.beginPath(); ctx.moveTo(x - 15, ly); ctx.lineTo(x + 15, ly); ctx.stroke(); } } } function drawNotes() { notes.forEach(n => { drawLedgerLines(n.x, n.y, n.index); ctx.beginPath(); ctx.arc(n.x, n.y, 6, 0, Math.PI * 2); ctx.fillStyle = "white"; ctx.fill(); // Stok aan de noot (optioneel) ctx.beginPath(); ctx.moveTo(n.x + 6, n.y); ctx.lineTo(n.x + 6, n.y - 30); ctx.stroke(); }); } function drawGhost() { if (!ghost) return; ctx.globalAlpha = 0.5; drawLedgerLines(ghost.x, ghost.y, ghost.index); ctx.beginPath(); ctx.arc(ghost.x, ghost.y, 6, 0, Math.PI * 2); ctx.fillStyle = "cyan"; // Ghost kleur ctx.fill(); ctx.globalAlpha = 1.0; } function loop() { ctx.clearRect(0, 0, canvas.width, canvas.height); drawStaff(); drawGhost(); drawNotes(); requestAnimationFrame(loop); } loop(); // Event Listeners (Unified voor muis en touch) function handleInput(e) { const rect = canvas.getBoundingClientRect(); const clientX = e.touches ? e.touches[0].clientX : e.clientX; const clientY = e.touches ? e.touches[0].clientY : e.clientY; // Offset correctie voor vinger-zichtbaarheid op touch const visualOffset = e.touches ? 40 : 0; return getNote(clientX - rect.left, clientY - rect.top - visualOffset); } canvas.addEventListener("pointerdown", (e) => { isTouching = true; ghost = handleInput(e); }); canvas.addEventListener("pointermove", (e) => { if (!isTouching) return; ghost = handleInput(e); }); window.addEventListener("pointerup", () => { if (isTouching && ghost) { notes.push(ghost); ghost = null; } isTouching = false; });
